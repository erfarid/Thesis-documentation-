' DOMAIN MODEL (ENTITIES)
'=========================================================
package "Domain Model (Entities)" {
  class Admin {
    +adminId: UUID
    +name: String
    +email: String
    +passwordHash: String
  }

  class Driver {
    +driverId: UUID
    +name: String
    +email: String
    +passwordHash: String
    +phone: String
  }

  class Vehicle {
    +vehicleId: UUID
    +plateNumber: String
    +ownerName: String
    +vehicleType: VehicleType
    +registrationStatus: RegistrationStatus
  }

  enum VehicleType {
    Car
    Van
    Motorcycle
    Truck
  }

  enum RegistrationStatus {
    Active
    Suspended
    Unknown
  }

  class ParkingZone {
    +zoneId: String
    +zoneName: String
    +baseHourlyRateHUF: int
    +latitude: double
    +longitude: double
  }

  class PeakHoursRule {
    +ruleId: UUID
    +startTime: Time
    +endTime: Time
    +daysOfWeek: String
    +multiplier: double
  }

  class SystemParameters {
    +maxAllowedDurationMinutes: int
    +overstayPenaltyMultiplier: double
    +repeatPenaltyIncrementPct: double
    +feeCapEnabled: bool
    +maxFeeCapHUF: int
    +yoloConfidenceThreshold: double
    +ocrConfidenceThreshold: double
    +plateRegex: String
  }

  class ParkingSession {
    +sessionId: String
    +entryTimestamp: DateTime
    +exitTimestamp: DateTime
    +durationMinutes: int
    +status: SessionStatus
  }

  enum SessionStatus {
    Active
    Closed
    Paid
    Unpaid
    Overdue
  }

  class FeeBreakdown {
    +baseFeeHUF: int
    +peakMultiplierApplied: double
    +overstayPenaltyHUF: int
    +repeatPenaltyHUF: int
    +capApplied: bool
    +finalFeeHUF: int
  }

  class Payment {
    +paymentId: UUID
    +amountHUF: int
    +paidAt: DateTime
    +method: PaymentMethod
    +status: PaymentStatus
  }

  enum PaymentMethod {
    Cash
    Card
    BankTransfer
    Simulated
  }

  enum PaymentStatus {
    Initiated
    Completed
    Failed
    Refunded
  }

  class ViolationRecord {
    +violationId: UUID
    +plateNumber: String
    +type: ViolationType
    +createdAt: DateTime
    +notes: String
  }

  enum ViolationType {
    Overstay
    Unpaid
    RepeatOffense
  }

  class RepeatOffenderProfile {
    +plateNumber: String
    +repeatCount: int
    +lastViolationAt: DateTime
  }

  class UploadedImage {
    +imageId: UUID
    +fileName: String
    +contentType: String
    +sizeBytes: long
    +uploadedAt: DateTime
  }

  class PlateRecognitionResult {
    +resultId: UUID
    +plateText: String
    +bboxX: int
    +bboxY: int
    +bboxW: int
    +bboxH: int
    +detectionConfidence: double
    +ocrConfidence: double
    +isValidFormat: bool
  }

  abstract class Report {
    +reportId: UUID
    +generatedAt: DateTime
    +dateFrom: Date
    +dateTo: Date
  }

  class RevenueReport {
    +totalRevenueHUF: long
    +paidCount: int
    +unpaidCount: int
    +overdueCount: int
  }

  class ViolationReport {
    +totalViolations: int
    +topRepeatOffenders: String
  }

  Report <|-- RevenueReport
  Report <|-- ViolationReport
}

' DOMAIN RELATIONS (AGGREGATION/COMPOSITION)
'=========================================================

' Driver owns vehicles (aggregation)
Driver "1" o-- "0..*" Vehicle : owns

' Vehicle has sessions (association)
Vehicle "1" -- "0..*" ParkingSession : used in

' Zone linked to sessions (association)
ParkingZone "1" -- "0..*" ParkingSession : occurs in

' Zone contains peak rules (composition)
ParkingZone "1" *-- "0..*" PeakHoursRule : peak rules

' Session contains fee breakdown (composition)
ParkingSession "1" *-- "0..1" FeeBreakdown : fee details

' Session contains payment (composition)
ParkingSession "1" *-- "0..1" Payment : payment

' Repeat offender profile aggregates violations (aggregation)
RepeatOffenderProfile "1" o-- "0..*" ViolationRecord : violation history

' Uploaded image produces recognition result (composition)
UploadedImage "1" *-- "0..1" PlateRecognitionResult : result

' Recognition result identifies a vehicle by plate (dependency)
PlateRecognitionResult ..> Vehicle : identifies by plate
